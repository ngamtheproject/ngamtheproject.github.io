{{- /* Gallery: every image in assets/media/albums//. Params: album (default "gallery"), order (asc|desc, by file name). Captions come from the page's `gallery_item` front matter: a list of {album, image, caption}. */ -}} {{- $album := .Get "album" | default "gallery" -}} {{- $order := .Get "order" | default "asc" -}} {{- $images := resources.Match (path.Join "media/albums" $album "*") -}} {{- if not $images }}{{ errorf "gallery: album %q is empty or missing at %s" $album .Position }}{{ end -}}
{{- range $i, $img := sort $images "Name" $order }} {{- $file := path.Base $img.Name -}} {{- $caption := "" -}} {{- range where (where ($.Page.Params.gallery_item | default slice) "album" $album) "image" $file }}{{ $caption = .caption }}{{ end }} {{- end }}
