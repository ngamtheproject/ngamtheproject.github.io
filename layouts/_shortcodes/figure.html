{{- /* Figure: an image from the page bundle or assets/media/, or a remote URL. Overrides Hugo's built-in `figure`, which would emit the raw `albums/...` path. Params: src, caption (alias: title), id, alt, link, class, max_width, width, height. */ -}} {{- $src := .Get "src" -}} {{- $caption := .Get "caption" | default (.Get "title") | default "" -}} {{- $id := anchorize (.Get "id" | default ($caption | plainify)) -}} {{- $alt := .Get "alt" | default ($caption | plainify) -}} {{- $link := .Get "link" | default "" -}} {{- $remote := strings.HasPrefix $src "http" -}} {{- $img := "" -}} {{- if not $remote -}} {{- $img = partial "media-image.html" (dict "page" .Page "path" $src) -}} {{- if not $img }}{{ errorf "figure: cannot find image %q (page bundle or assets/media/) at %s" $src .Position }}{{ end -}} {{- end -}}
