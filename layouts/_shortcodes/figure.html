{{/* Figure shortcode (ported from HugoBlox Bootstrap v5 for HugoBlox Kit). Loads the image from the page bundle, falling back to the media library at `assets/media/`, then to a remote URL. Overrides Hugo's built-in `figure`, which would emit the raw `albums/...` path. Parameters: src, caption (legacy alias: title), id, alt, link, class, max_width, width, height, theme (light|dark). */}} {{- $destination := .Get "src" -}} {{- $is_remote := strings.HasPrefix $destination "http" -}} {{- $caption := .Get "caption" | default (.Get "title") | default "" -}} {{- $id := anchorize (.Get "id" | default ($caption | plainify)) -}} {{- $alt := .Get "alt" | default ($caption | plainify) -}} {{- $link := .Get "link" | default "" -}} {{- $img_class := "" -}} {{- if eq (.Get "theme" | lower) "light" }}{{ $img_class = "img-light" }}{{ end -}} {{- if eq (.Get "theme" | lower) "dark" }}{{ $img_class = "img-dark" }}{{ end -}} {{- $img := "" -}} {{- if not $is_remote -}} {{- $img = (.Page.Resources.ByType "image").GetMatch $destination -}} {{- if not $img -}} {{- $img = resources.Get (path.Join "media" $destination) -}} {{- end -}} {{- if not $img -}} {{- errorf "figure: cannot find image %q (looked in the page bundle and assets/media/) in %s" $destination .Position -}} {{- end -}} {{- end -}}
{{- if $link }}{{ end -}} {{- if $img -}} {{- $canProcess := and (reflect.IsImageResourceProcessable $img) (ne $img.MediaType.SubType "gif") -}} {{- if not $canProcess -}} {{- else -}} {{- $img_lg := $img.Fit "1200x1200 webp" -}} {{- $img_md := $img_lg.Fit "760x760 webp" -}} {{- $img_sm := $img_md.Fit "400x400 webp" -}} {{- end -}} {{- else -}} {{- end -}} {{- if $link }}{{ end -}}
